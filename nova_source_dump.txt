#include "action_parser.hpp"

using namespace std;

ActionParserResult ActionParser::parse(
    const string& input
)
{
    ActionParserResult result;

    if (input.empty())
    {
        result.success = false;
        result.action.name = "";
        result.message =
            "Cannot parse an empty action request.";

        return result;
    }

    // --------------------------------------------------------
    // ACTION — CREATE FOLDER
    // --------------------------------------------------------

    if (
        input.find("create a folder") != string::npos ||
        input.find("make a folder") != string::npos
    )
    {
        result.success = true;

        result.action.name =
            "create_folder";

        result.message =
            "Action parsed as create_folder.";

        return result;
    }

    // --------------------------------------------------------
    // ACTION — OPEN APPLICATION
    // --------------------------------------------------------

    if (
        input.find("open ") == 0 ||
        input.find("launch ") == 0
    )
    {
        result.success = true;

        result.action.name =
            "open_app";

        result.message =
            "Action parsed as open_app.";

        return result;
    }

    // --------------------------------------------------------
    // ACTION — CLOSE APPLICATION
    // --------------------------------------------------------

    if (
        input.find("close ") == 0
    )
    {
        result.success = true;

        result.action.name =
            "close_app";

        result.message =
            "Action parsed as close_app.";

        return result;
    }

    // --------------------------------------------------------
    // UNKNOWN ACTION
    // --------------------------------------------------------

    result.success = false;

    result.action.name = "";

    result.message =
        "Unable to determine requested action.";

    return result;
}

namespace

    string toLower(string text)
    {
        transform(
            text.begin(),
            text.end(),
            text.begin(),
            [](unsigned char character)
            {
                return static_cast<char>(
                    tolower(character)
                );
            }
        );

        return text;
    }

    bool contains(
        const string& text,
        const string& value
    )
    {
        return text.find(value) != string::npos;
    }

    PrimitiveOperation makePrimitive(
        PrimitiveOperationType type,
        const string& name
    )
    {
        PrimitiveOperation operation;

        operation.type = type;
        operation.name = name;
        operation.parameters =
        PrimitiveParameters{};
        operation.resolved = true;

        return operation;
    }

    string buildPrimitiveList()
    {
        const vector<string> primitives =
            PrimitiveRegistry::getSupportedNames();

        ostringstream output;

        for (const auto& primitive : primitives)
        {
            output << "- "
                   << primitive
                   << "\n";
        }

        return output.str();
    }

    bool addValidatedPrimitive(
        MultiStepAction& action,
        const string& primitiveName,
        string& errorMessage
    )
    {
        PrimitiveOperationType type;

        if (!PrimitiveRegistry::resolveType(
                primitiveName,
                type))
        {
            errorMessage =
                "LLM proposed unsupported primitive: " +
                primitiveName;

            return false;
        }

        action.operations.push_back(
            makePrimitive(
                type,
                primitiveName
            )
        );

        return true;
    }
}

ActionPrimitiveResolutionResult
ActionPrimitiveResolver::resolve(
    const MultiStepAction& action
)
{
    ActionPrimitiveResolutionResult result;

    result.status =
        ActionPrimitiveResolutionStatus::FAILED;

    result.action = action;
    result.message = "";
    result.clarificationQuestion = "";

    // --------------------------------------------------
    // INPUT VALIDATION
    // --------------------------------------------------

    if (action.id.empty())
    {
        result.message =
            "Cannot resolve action without an ID.";

        return result;
    }

    if (action.description.empty())
    {
        result.message =
            "Cannot resolve action without a description.";

        return result;
    }

    const string text =
        toLower(action.description);

    MultiStepAction resolvedAction =
        action;

    resolvedAction.operations.clear();
    // --------------------------------------------------
    // DETERMINISTIC: FILESYSTEM CREATE
    // --------------------------------------------------

    if (
        contains(text, "create")
        ||
        contains(text, "add")
    )
    {
        size_t createPosition =
            text.find("create");

        if (createPosition == string::npos)
        {
            createPosition =
                text.find("add");
        }

        size_t directoryPosition =
            text.find(
                " directory",
                createPosition
            );
        size_t projectPosition =
            text.find(
                " project",
                createPosition
            );

        size_t folderPosition =
            text.find(
                " folder",
                createPosition
            );

        size_t filePosition =
            text.find(
                " file",
                createPosition
            );

        size_t javaPosition =
            text.find(
                ".java",
                createPosition
            );

        size_t cppPosition =
            text.find(
                ".cpp",
                createPosition
            );

        size_t pythonPosition =
            text.find(
                ".py",
                createPosition
            );

        size_t txtPosition =
            text.find(
                ".txt",
                createPosition
            );

        // --------------------------------------------------
        // Determine which object appears first after "create"
        // --------------------------------------------------
        size_t directoryObjectPosition =
            string::npos;

        if (directoryPosition != string::npos)
        {
            directoryObjectPosition =
                directoryPosition;
        }

        if (
            projectPosition != string::npos
            &&
            (
                directoryObjectPosition == string::npos
                ||
                projectPosition < directoryObjectPosition
            )
        )
        {
            directoryObjectPosition =
                projectPosition;
        }

        if (
            folderPosition != string::npos
            &&
            (
                directoryObjectPosition == string::npos
                ||
                folderPosition < directoryObjectPosition
            )
        )
        {
            directoryObjectPosition =
                folderPosition;
        }

        size_t fileObjectPosition =
            string::npos;

        if (filePosition != string::npos)
        {
            fileObjectPosition =
                filePosition;
        }

        const size_t extensionPositions[] =
        {
            javaPosition,
            cppPosition,
            pythonPosition,
            txtPosition
        };

        for (size_t position : extensionPositions)
        {
            if (
                position != string::npos
                &&
                (
                    fileObjectPosition == string::npos
                    ||
                    position < fileObjectPosition
                )
            )
            {
                fileObjectPosition =
                    position;
            }
        }

        // --------------------------------------------------
        // CREATE DIRECTORY / FOLDER
        // --------------------------------------------------

        if (
            directoryObjectPosition !=
            string::npos
            &&
            (
                fileObjectPosition ==
                string::npos
                ||
                directoryObjectPosition <
                fileObjectPosition
            )
        )
        {
            resolvedAction.operations.push_back(
                makePrimitive(
                    PrimitiveOperationType::FILESYSTEM_CREATE,
                    "filesystem.create"
                )
            );

            resolvedAction.operation =
                "filesystem.create";

            resolvedAction.capability =
                "filesystem";

            resolvedAction.resolved =
                true;

            result.status =
                ActionPrimitiveResolutionStatus::RESOLVED;

            result.action =
                resolvedAction;

            result.message =
                "Resolved deterministically to filesystem.create.";

            return result;
        }

        // --------------------------------------------------
        // CREATE FILE
        // --------------------------------------------------

        if (
            fileObjectPosition !=
            string::npos
        )
        {
            resolvedAction.operations.push_back(
                makePrimitive(
                    PrimitiveOperationType::FILESYSTEM_CREATE_FILE,
                    "filesystem.create_file"
                )
            );

            resolvedAction.operation =
                "filesystem.create_file";

            resolvedAction.capability =
                "filesystem";

            resolvedAction.resolved =
                true;

            result.status =
                ActionPrimitiveResolutionStatus::RESOLVED;

            result.action =
                resolvedAction;

            result.message =
                "Resolved deterministically to filesystem.create_file.";

            return result;
        }
    }

    
    // --------------------------------------------------
    // DETERMINISTIC: APPLICATION LAUNCH
    // --------------------------------------------------

    if (
        contains(text, "open")
        ||
        contains(text, "launch")
        ||
        contains(text, "start")
    )
    {
        resolvedAction.operations.push_back(
            makePrimitive(
                PrimitiveOperationType::APPLICATION_LAUNCH,
                "application.launch"
            )
        );

        resolvedAction.operation =
            "application.launch";

        resolvedAction.capability =
            "application_control";

        resolvedAction.resolved = true;

        result.status =
            ActionPrimitiveResolutionStatus::RESOLVED;

        result.action =
            resolvedAction;

        result.message =
            "Resolved deterministically to application.launch.";

        return result;
    }

    // --------------------------------------------------
    // DETERMINISTIC: TEXT WRITE
    // --------------------------------------------------

    if (
        contains(text, "write")
        ||
        contains(text, "type")
        ||
        contains(text, "enter")
    )
    {
        resolvedAction.operations.push_back(
            makePrimitive(
                PrimitiveOperationType::TEXT_WRITE,
                "text.write"
            )
        );

        resolvedAction.operation =
            "text.write";

        resolvedAction.capability =
            "text_input";

        resolvedAction.resolved = true;

        result.status =
            ActionPrimitiveResolutionStatus::RESOLVED;

        result.action =
            resolvedAction;

        result.message =
            "Resolved deterministically to text.write.";

        return result;
    }

    // --------------------------------------------------
    // LLM FALLBACK
    // --------------------------------------------------

    const string systemPrompt =
        R"(
You are Nova's Action / Primitive Resolver.

Your job is to convert ONE logical action step into
one or more primitive operations.

You are NOT allowed to execute anything.

You are NOT allowed to invent primitives.

You may ONLY use primitives from the supplied registry.

CRITICAL RULE:

If the logical step is incomplete, ambiguous, or does not
contain enough information to determine the required primitive
operations without inventing missing facts, DO NOT guess.

Instead return:

{
  "status": "clarification_required",
  "question": "A concise question that Nova should ask the user."
}

For a sufficiently clear step, return:

{
  "status": "resolved",
  "operations": [
    {
      "name": "primitive.name"
    }
  ]
}

Rules:

- Use the smallest meaningful primitive sequence.
- Do not add redundant operations.
- Do not repeat the same primitive unless the logical step
  genuinely requires multiple uses.
- Preserve the meaning of the logical step.
- Do not resolve concrete filesystem paths.
- Do not invent application names.
- Do not provide shell commands.
- Do not provide explanations outside JSON.
- Every operation name MUST come from the supplied primitive registry.
- If information is missing, ask for clarification instead of guessing.
)";

    const string userPrompt =
        "Logical step:\n\n" +
        action.description +
        "\n\n" +
        "Available primitives:\n" +
        buildPrimitiveList();

    LLMClient client;

    LLMResponse llmResponse =
        client.generate(
            systemPrompt,
            userPrompt
        );

    if (!llmResponse.success)
    {
        result.message =
            "Primitive resolution LLM request failed: " +
            llmResponse.message;

        return result;
    }

    if (llmResponse.content.empty())
    {
        result.message =
            "Primitive resolution LLM returned empty content.";

        return result;
    }

    // --------------------------------------------------
    // PARSE LLM RESPONSE
    // --------------------------------------------------

    try
    {
        json parsed =
            json::parse(
                llmResponse.content
            );

        if (
            !parsed.contains("status")
            ||
            !parsed["status"].is_string()
        )
        {
            result.message =
                "LLM primitive response does not contain a valid status.";

            return result;
        }

        const string status =
            parsed["status"].get<string>();

        // --------------------------------------------------
        // CLARIFICATION REQUIRED
        // --------------------------------------------------

        if (status == "clarification_required")
        {
            if (
                !parsed.contains("question")
                ||
                !parsed["question"].is_string()
            )
            {
                result.message =
                    "LLM requested clarification without providing a question.";

                return result;
            }

            result.status =
                ActionPrimitiveResolutionStatus::CLARIFICATION_REQUIRED;

            result.clarificationQuestion =
                parsed["question"].get<string>();

            result.message =
                "Action requires user clarification.";

            return result;
        }

        // --------------------------------------------------
        // RESOLVED
        // --------------------------------------------------

        if (status != "resolved")
        {
            result.message =
                "LLM returned an unsupported resolution status: " +
                status;

            return result;
        }

        if (
            !parsed.contains("operations")
            ||
            !parsed["operations"].is_array()
        )
        {
            result.message =
                "Resolved LLM response does not contain a valid operations array.";

            return result;
        }

        if (parsed["operations"].empty())
        {
            result.message =
                "Resolved LLM response contains no operations.";

            return result;
        }

        for (
            const auto& operationJson :
            parsed["operations"]
        )
        {
            if (
                !operationJson.contains("name")
                ||
                !operationJson["name"].is_string()
            )
            {
                result.message =
                    "LLM primitive response contains an operation without a valid name.";

                result.action.operations.clear();

                return result;
            }

            const string primitiveName =
                operationJson["name"].get<string>();

            string validationError;

            if (!addValidatedPrimitive(
                    resolvedAction,
                    primitiveName,
                    validationError))
            {
                result.message =
                    validationError;

                result.action.operations.clear();

                return result;
            }
        }

        resolvedAction.resolved = true;

        result.status =
            ActionPrimitiveResolutionStatus::RESOLVED;

        result.action =
            resolvedAction;

        result.message =
            "Resolved through focused LLM primitive composition.";

        return result;
    }
    catch (const json::exception& error)
    {
        result.message =
            string(
                "Invalid JSON from primitive resolver LLM: "
            ) +
            error.what();

        result.action.operations.clear();

        return result;
    }
}

ActionValidationResult ActionValidator::validate(
    const Action& action,
    const Capability& capability,
    const ActionParameters& parameters
)
{
    ActionValidationResult result;

    result.valid = false;
    result.message = "";

    // --------------------------------------------------------
    // BASIC VALIDATION
    // --------------------------------------------------------

    if (action.name.empty())
    {
        result.message =
            "Action is missing.";

        return result;
    }

    if (!capability.supported)
    {
        result.message =
            "Capability is unsupported.";

        return result;
    }

    // --------------------------------------------------------
    // CREATE FOLDER
    // --------------------------------------------------------

    if (
        action.name ==
        "create_folder"
    )
    {
        if (parameters.name.type != ParameterValueType::LITERAL ||
            parameters.name.value.empty())
        {
            result.message =
                "Folder name is required.";

            return result;
        }

        if (parameters.location.type != ParameterValueType::LITERAL ||
            parameters.location.value.empty())
        {
            result.message =
                "Folder location is required.";

            return result;
        }

        result.valid = true;

        result.message =
            "Create folder action is valid.";

        return result;
    }

    // --------------------------------------------------------
    // OPEN / CLOSE APPLICATION
    // --------------------------------------------------------

    if (
        action.name ==
        "open_app" ||
        action.name ==
        "close_app"
    )
    {
        if (parameters.application.type != ParameterValueType::LITERAL ||
            parameters.application.value.empty())
        {
            result.message =
                "Application parameter is required.";

            return result;
        }

        result.valid = true;

        result.message =
            "Application action is valid.";

        return result;
    }

    // --------------------------------------------------------
    // UNKNOWN ACTION
    // --------------------------------------------------------

    result.message =
        "No validation rule exists for requested action.";

    return result;
}

#include <windows.h>

#include "application_resolver.hpp"

#include <algorithm>
#include <cctype>
#include <cstdio>

namespace
{
    std::string normalize(
        const std::string& value
    )
    {
        std::string result = value;

        std::transform(
            result.begin(),
            result.end(),
            result.begin(),
            [](
                unsigned char character
            )
            {
                return static_cast<char>(
                    std::tolower(character)
                );
            }
        );

        return result;
    }

    std::string escapePowerShell(
        const std::string& value
    )
    {
        std::string result;

        for (char character : value)
        {
            if (character == '\'')
            {
                result += "''";
            }
            else
            {
                result += character;
            }
        }

        return result;
    }
}

ApplicationResolverResult ApplicationResolver::resolve(
    const std::string& application
)
{
    ApplicationResolverResult result{};

    result.success = false;
    result.application = application;
    result.appId = "";
    result.displayName = "";
    result.message = "";

    if (application.empty())
    {
        result.message =
            "Application name is empty.";

        return result;
    }

    const std::string normalized =
        normalize(application);

    const std::string escaped =
        escapePowerShell(application);

    const std::string command =
        "powershell.exe -NoProfile -Command "
        "\""
        "$app = Get-StartApps | "
        "Where-Object { "
        "$_.Name -like '*"
        + escaped +
        "*' "
        "-or "
        "$_.AppID -like '*"
        + escaped +
        "*' "
        "} | Select-Object -First 1; "
        "if ($null -ne $app) { "
        "[Console]::WriteLine($app.Name); "
        "[Console]::WriteLine($app.AppID); "
        "}"
        "\"";

    FILE* pipe =
        _popen(
            command.c_str(),
            "r"
        );

    if (pipe == nullptr)
    {
        result.message =
            "Failed to query Windows Start Apps.";

        return result;
    }

    char buffer[1024];

    std::string output;

    while (
        fgets(
            buffer,
            sizeof(buffer),
            pipe
        ) != nullptr
    )
    {
        output += buffer;
    }

    const int exitCode =
        _pclose(pipe);

    if (exitCode != 0 || output.empty())
    {
        result.message =
            "Windows Start Apps could not resolve application: "
            + application;

        return result;
    }

    const std::size_t separator =
        output.find('\n');

    if (separator == std::string::npos)
    {
        result.message =
            "Windows returned an invalid application result.";

        return result;
    }

    result.displayName =
        output.substr(
            0,
            separator
        );

    result.appId =
        output.substr(
            separator + 1
        );

    while (
        !result.displayName.empty()
        &&
        (
            result.displayName.back() == '\r'
            ||
            result.displayName.back() == '\n'
        )
    )
    {
        result.displayName.pop_back();
    }

    while (
        !result.appId.empty()
        &&
        (
            result.appId.back() == '\r'
            ||
            result.appId.back() == '\n'
        )
    )
    {
        result.appId.pop_back();
    }

    if (
        result.displayName.empty()
        ||
        result.appId.empty()
    )
    {
        result.message =
            "Windows returned an incomplete application result.";

        return result;
    }

    result.success = true;

    result.message =
        "Application resolved through Windows Start Apps.";

    return result;
}

ContentResolutionResult ContentResolver::resolve(
    const string& description,

{

    result.status =
        ContentResolutionStatus::FAILED;

    result.content = "";
    result.message = "";
    result.clarificationQuestion = "";

    if (description.empty())
    {
        result.message =
            "Cannot resolve content from empty description.";

        return result;
    }

    if (target.empty())
    {
        result.message =
            "Cannot resolve content without a target.";

        return result;
    }

    const string systemPrompt =
        R"(
You are Nova's Content Resolver.

Your job is to convert a logical content instruction into
the actual content that should be written to the specified file.

You are NOT allowed to execute anything.

You are NOT allowed to modify files.

You must return ONLY valid JSON.

For a sufficiently clear instruction, return:

{
  "status": "resolved",
  "content": "the exact content to write"
}

If the instruction is incomplete or ambiguous and you cannot
determine the intended content without inventing user intent,
return:

{
  "status": "clarification_required",
  "question": "A concise question Nova should ask the user."
}

Rules:

- Preserve the requested intent.
- Generate the actual file content, not a description of it.
- Do not include Markdown fences.
- Do not include explanations outside JSON.
- Do not provide shell commands.
- Do not execute anything.
- Do not invent unrelated functionality.
- Keep generated content minimal and appropriate to the request.
)";

    const string userPrompt =
        "Target file:\n" +
        target +
        "\n\n" +
        "Content instruction:\n" +
        description;

    LLMClient client;

    LLMResponse llmResponse =
        client.generate(
            systemPrompt,
            userPrompt
        );

    if (!llmResponse.success)
    {
        result.message =
            "Content resolution LLM request failed: " +
            llmResponse.message;

        return result;
    }

    if (llmResponse.content.empty())
    {
        result.message =
            "Content resolution LLM returned empty content.";

        return result;
    }

    try
    {
        json parsed =
            json::parse(
                llmResponse.content
            );

        if (
            !parsed.contains("status")
            ||
            !parsed["status"].is_string()
        )
        {
            result.message =
                "Content resolver response does not contain a valid status.";

            return result;
        }

        const string status =
            parsed["status"].get<string>();

        if (status == "clarification_required")
        {
            if (
                !parsed.contains("question")
                ||
                !parsed["question"].is_string()
            )
            {
                result.message =
                    "Content resolver requested clarification without providing a question.";

                return result;
            }

            result.status =
                ContentResolutionStatus::CLARIFICATION_REQUIRED;

            result.clarificationQuestion =
                parsed["question"].get<string>();

            result.message =
                "Content resolution requires clarification.";

            return result;
        }

        if (status != "resolved")
        {
            result.message =
                "Content resolver returned an unknown status.";

            return result;
        }

        if (
            !parsed.contains("content")
            ||
            !parsed["content"].is_string()
        )
        {
            result.message =
                "Content resolver returned resolved status without valid content.";

            return result;
        }

        result.content =
            parsed["content"].get<string>();

        if (result.content.empty())
        {
            result.message =
                "Content resolver returned empty content.";

            return result;
        }

        result.status =
            ContentResolutionStatus::RESOLVED;

        result.message =
            "Content resolved successfully.";

        return result;
    }
    catch (const exception& exception)
    {
        result.message =
            string("Failed to parse content resolver response: ") +
            exception.what();

        return result;
    }

    bool resolveOutput(
        const MultiStepAction& sourceStep,
        const string& outputName,
        string& value
    )
    {
        for (const PrimitiveOperation& operation : sourceStep.operations)
        {
            const PrimitiveParameters& parameters = operation.parameters;

            if (outputName == "name" && !parameters.name.empty())
            {
                value = parameters.name;
                return true;
            }
            if (outputName == "location" && !parameters.location.empty())
            {
                value = parameters.location;
                return true;
            }
            if (outputName == "application" && !parameters.application.empty())
            {
                value = parameters.application;
                return true;
            }
            if (outputName == "target" && !parameters.target.empty())
            {
                value = parameters.target;
                return true;
            }
            if (outputName == "content" && !parameters.content.empty())
            {
                value = parameters.content;
                return true;
            }
            if (outputName == "path")
            {
                if (!parameters.location.empty() &&
                    !parameters.name.empty())
                {
                    value = parameters.location + "/" + parameters.name;
                    return true;
                }
                if (!parameters.target.empty())
                {
                    value = parameters.target;
                    return true;
                }
            }
        }

        return false;
    }

    ParameterValue makeLiteral(const string& value)
    {
        return ParameterValue{
            ParameterValueType::LITERAL,
            value,
            ContextParameterReference{}
        };
    }

    bool convertToPrimitiveParameters(
        const ActionParameters& parameters,
        PrimitiveParameters& primitiveParameters,
        string& errorMessage
    )
    {
        const auto convert = [&errorMessage](
            const ParameterValue& source,
            string& destination,
            const string& name)
        {
            if (source.type == ParameterValueType::NONE)
            {
                destination.clear();
                return true;
            }

            if (source.type != ParameterValueType::LITERAL ||
                source.value.empty())
            {
                errorMessage =
                    "Parameter is not a concrete execution value: " + name;
                return false;
            }

            destination = source.value;
            return true;
        };

        return
            convert(parameters.name, primitiveParameters.name, "name") &&
            convert(parameters.location, primitiveParameters.location, "location") &&
            convert(parameters.application, primitiveParameters.application, "application") &&
            convert(parameters.target, primitiveParameters.target, "target") &&
            convert(parameters.content, primitiveParameters.content, "content");
    }

ContextResolutionResult ContextResolver::resolve(
    const MultiStepAction& action,
    const MultiStepPlan& plan
)
{
    if (!action.semanticParameters.empty())
    {
        ContextResolutionResult semanticResult;
        semanticResult.action = action;

        if (action.semanticParameters.size() != action.operations.size())
        {
            semanticResult.status = ContextResolutionStatus::FAILED;
            semanticResult.message =
                "Semantic parameter count does not match primitive count.";
            return semanticResult;
        }

        size_t currentIndex = plan.steps.size();
        for (size_t index = 0; index < plan.steps.size(); ++index)
        {
            if (plan.steps[index].id == action.id)
            {
                currentIndex = index;
                break;
            }
        }

        MultiStepAction resolvedAction = action;
        for (size_t operationIndex = 0;
             operationIndex < resolvedAction.operations.size();
             ++operationIndex)
        {
            ActionParameters& parameters =
                resolvedAction.semanticParameters[operationIndex];

            const auto resolveParameter = [&](ParameterValue& parameter)
            {
                if (parameter.type != ParameterValueType::CONTEXT_REFERENCE)
                {
                    return true;
                }

                size_t sourceIndex = plan.steps.size();
                const MultiStepAction* sourceStep = nullptr;
                for (size_t index = 0; index < plan.steps.size(); ++index)
                {
                    if (plan.steps[index].id == parameter.reference.stepId)
                    {
                        sourceIndex = index;
                        sourceStep = &plan.steps[index];
                        break;
                    }
                }

                string resolvedValue;
                if (currentIndex == plan.steps.size() ||
                    sourceIndex >= currentIndex ||
                    sourceStep == nullptr ||
                    !resolveOutput(
                        *sourceStep,
                        parameter.reference.output,
                        resolvedValue))
                {
                    return false;
                }

                const string referenceName =
                    parameter.reference.stepId + "." +
                    parameter.reference.output;
                resolvedAction.contextReferences.push_back(
                    ContextReference{
                        ContextReferenceType::PREVIOUS_STEP_OUTPUT,
                        referenceName,
                        parameter.reference.stepId,
                        true,
                        resolvedValue
                    });
                semanticResult.resolvedContext[referenceName] = resolvedValue;
                parameter = makeLiteral(resolvedValue);
                return true;
            };

            if (!resolveParameter(parameters.name) ||
                !resolveParameter(parameters.location) ||
                !resolveParameter(parameters.application) ||
                !resolveParameter(parameters.target) ||
                !resolveParameter(parameters.content))
            {
                semanticResult.status =
                    ContextResolutionStatus::CLARIFICATION_REQUIRED;
                semanticResult.clarificationQuestion =
                    "A referenced previous-step output could not be resolved.";
                semanticResult.message =
                    "Semantic context reference could not be resolved.";
                return semanticResult;
            }

            PrimitiveParameters primitiveParameters;
            string conversionError;
                if (!convertToPrimitiveParameters(
                    parameters,
                    primitiveParameters,
                    conversionError))
            {
                semanticResult.status = ContextResolutionStatus::FAILED;
                semanticResult.message = conversionError;
                return semanticResult;
            }

            resolvedAction.operations[operationIndex].parameters =
                primitiveParameters;
        }

        semanticResult.status = ContextResolutionStatus::RESOLVED;
        semanticResult.action = resolvedAction;
        semanticResult.message = "Semantic parameters resolved successfully.";
        return semanticResult;
    }

    ContextResolutionResult result;

    result.action =
        action;

    // --------------------------------------------------------
    // NO CONTEXT REFERENCE
    // --------------------------------------------------------

    if (
        action.dependencies.empty() &&
        !containsContextReference(action.description)
    )
    {
        result.status = ContextResolutionStatus::RESOLVED;
        result.message =
            "No contextual references require resolution.";
        return result;
    }

    // --------------------------------------------------------
    // RESOLVE PREVIOUS-STEP CONTEXT
    // --------------------------------------------------------

    MultiStepAction resolvedAction =
        action;

    bool resolved =
        resolvePreviousStepContext(
            resolvedAction,
            plan
        );

    if (resolved)
    {
        result.status =
            ContextResolutionStatus::RESOLVED;

        result.action =
            resolvedAction;

        for (
            const ContextReference& reference
            : resolvedAction.contextReferences
        )
        {
            if (reference.resolved)
            {
                result.resolvedContext[
                    reference.reference
                ] =
                    reference.resolvedValue;

                result.resolvedContext[action.description] =
                    reference.resolvedValue;
            }
        }

        result.message =
            "Context resolved successfully.";

        return result;
    }

    // --------------------------------------------------------
    // CONTEXT COULD NOT BE RESOLVED
    // --------------------------------------------------------

    result.status =
        ContextResolutionStatus::CLARIFICATION_REQUIRED;

    result.clarificationQuestion =
        "I could not determine which previous resource this reference refers to.";

    result.message =
        "Context reference could not be resolved.";

    return result;
}

bool ContextResolver::containsContextReference(
    const string& description
)
{
    const string text = description;

    return
        text.find("that folder") != string::npos ||
        text.find("the folder") != string::npos ||
        text.find("that file") != string::npos ||
        text.find("the file") != string::npos ||
        text.find("inside the ") != string::npos ||
        text.find("within the ") != string::npos ||
        text.find("in the project directory") != string::npos ||
        text.find("project directory in ") != string::npos ||
        text.find(" project in ") != string::npos;
}

    string toLower(string text)
    {
        transform(
            text.begin(),
            text.end(),
            text.begin(),
            [](unsigned char character)
            {
                return static_cast<char>(
                    tolower(character)
                );
            }
        );

        return text;
    }

    bool contains(
        const string& text,
        const string& value
    )
    {
        return text.find(value) != string::npos;
    }

    bool referencesStepResource(
        const MultiStepAction& currentStep,
        const MultiStepAction& previousStep
    )
    {
        const string current =
            toLower(currentStep.description);

        const string resource =
            toLower(previousStep.description);

        /*
         * Resource names are inferred from the already-resolved
         * primitive parameters when available.
         */

        for (const auto& operation :
             previousStep.operations)
        {
            const string name =
                toLower(operation.parameters.name);

            if (
                !name.empty()
                &&
                contains(current, name)
            )
            {
                return true;
            }
        }

        /*
         * Explicit contextual language can also indicate
         * dependency on the previous step's resource.
         */

        if (
            contains(current, "inside the")
            ||
            contains(current, "inside ")
            ||
            contains(current, "within the")
            ||
            contains(current, "in the")
            ||
            contains(current, "open the")
            ||
            contains(current, "that folder")
            ||
            contains(current, "that directory")
            ||
            contains(current, "that file")
        )
        {
            return true;
        }

        return false;
    }
}

DependencyResolutionResult DependencyResolver::resolve(
    const MultiStepPlan& plan
)
{
    DependencyResolutionResult result;

    result.valid = false;
    result.executionOrder.clear();
    result.message = "";

    MultiStepPlan resolvedPlan = plan;

    if (!plan.valid)
    {
        result.message =
            "Cannot resolve dependencies from an invalid multi-step plan.";

        return result;
    }

    if (plan.steps.empty())
    {
        result.message =
            "Multi-step plan contains no steps.";

        return result;
    }

    unordered_map<string, const MultiStepAction*> stepMap;

    for (const auto& step : plan.steps)
    {
        if (step.id.empty())
        {
            result.message =
                "A step has an empty ID.";

            return result;
        }

        if (stepMap.contains(step.id))
        {
            result.message =
                "Duplicate step ID: " + step.id;

            return result;
        }

        stepMap[step.id] = &step;
    }

    // --------------------------------------------------
    // INFER MISSING RESOURCE DEPENDENCIES
    // --------------------------------------------------

    for (size_t currentIndex = 0;
        currentIndex < resolvedPlan.steps.size();
        ++currentIndex)
    {
        auto& currentStep =
            resolvedPlan.steps[currentIndex];

        for (size_t previousIndex = 0;
            previousIndex < currentIndex;
            ++previousIndex)
        {
            const auto& previousStep =
                resolvedPlan.steps[previousIndex];

            if (
                referencesStepResource(
                    currentStep,
                    previousStep
                )
            )
            {
                if (
                    find(
                        currentStep.dependencies.begin(),
                        currentStep.dependencies.end(),
                        previousStep.id
                    )
                    ==
                    currentStep.dependencies.end()
                )
                {
                    currentStep.dependencies.push_back(
                        previousStep.id
                    );
                }
            }
        }
    }

    unordered_map<string, int> inDegree;

    unordered_map<string, vector<string>> dependents;

    for (const auto& step : resolvedPlan.steps)
    {
        inDegree[step.id] = 0;
    }

    for (const auto& step : resolvedPlan.steps)
    {
        for (const auto& dependency : step.dependencies)
        {
            if (!stepMap.contains(dependency))
            {
                result.message =
                    "Step '" +
                    step.id +
                    "' depends on unknown step '" +
                    dependency +
                    "'.";

                return result;
            }

            if (dependency == step.id)
            {
                result.message =
                    "Step '" +
                    step.id +
                    "' cannot depend on itself.";

                return result;
            }

            dependents[dependency].push_back(step.id);

            inDegree[step.id]++;
        }
    }

    queue<string> readySteps;

    for (const auto& step : resolvedPlan.steps)
    {
        if (inDegree[step.id] == 0)
        {
            readySteps.push(step.id);
        }
    }

    while (!readySteps.empty())
    {
        string current =
            readySteps.front();

        readySteps.pop();

        result.executionOrder.push_back(current);

        for (const auto& dependent :
             dependents[current])
        {
            inDegree[dependent]--;

            if (inDegree[dependent] == 0)
            {
                readySteps.push(dependent);
            }
        }
    }

    if (
        result.executionOrder.size() !=
        resolvedPlan.steps.size()
    )
    {
        result.executionOrder.clear();

        result.message =
            "Dependency cycle detected in multi-step plan.";

        return result;
    }

    result.plan =
        resolvedPlan;

    result.valid = true;

    result.message =
        "Dependencies resolved successfully.";

    return result;
}

    }

    // --------------------------------------------------

        !plan.parameters.location.empty()

    )
    {
        result.confidence += 0.10;

    // --------------------------------------------------

ExecutionPlan ExecutionPlanBuilder::build(
    const Action& action,
    const Capability& capability,
    const ActionParameters& parameters,
    const ActionValidationResult& validation
)
{
    ExecutionPlan plan;

    plan.valid = false;
    plan.type = "";
    plan.action.name = "";
    plan.capability.supported = false;
    plan.capability.name = "";
    plan.capability.message = "";

    plan.parameters = PrimitiveParameters{};

    plan.message = "";

    // --------------------------------------------------------
    // VALIDATION GATE
    // --------------------------------------------------------

    if (!validation.valid)
    {
        plan.message =
            "Cannot build execution plan from invalid action.";

        return plan;
    }

    // --------------------------------------------------------
    // BUILD PLAN
    // --------------------------------------------------------

    plan.type =
        "single_action";

    plan.action =
        action;

    plan.capability =
        capability;

    const auto copyLiteral = [](
        const ParameterValue& source,
        string& destination)
    {
        if (source.type == ParameterValueType::NONE)
        {
            destination.clear();
            return true;
        }

        if (source.type != ParameterValueType::LITERAL ||
            source.value.empty())
        {
            return false;
        }

        destination = source.value;
        return true;
    };

    if (!copyLiteral(parameters.name, plan.parameters.name) ||
        !copyLiteral(parameters.location, plan.parameters.location) ||
        !copyLiteral(parameters.application, plan.parameters.application) ||
        !copyLiteral(parameters.target, plan.parameters.target) ||
        !copyLiteral(parameters.content, plan.parameters.content))
    {
        plan.message =
            "Cannot build execution plan with unresolved semantic parameters.";
        return plan;
    }

    plan.valid = true;

    plan.message =
        "Execution plan built successfully.";
    return plan;
}

ExecutionPlan ExecutionPlanBuilder::build(
    const MultiStepPlan& plan,
    const vector<string>& executionOrder
)
{
    ExecutionPlan executionPlan;

    executionPlan.valid = false;
    executionPlan.type = "multi_step";
    executionPlan.steps.clear();
    executionPlan.message = "";

    // --------------------------------------------------------
    // INPUT VALIDATION
    // --------------------------------------------------------

    if (!plan.valid)
    {
        executionPlan.message =
            "Cannot build execution plan from invalid multi-step plan.";

        return executionPlan;
    }

    if (plan.steps.empty())
    {
        executionPlan.message =
            "Cannot build execution plan from empty multi-step plan.";

        return executionPlan;
    }

    if (executionOrder.empty())
    {
        executionPlan.message =
            "Cannot build execution plan without execution order.";

        return executionPlan;
    }

    if (
        executionOrder.size() !=
        plan.steps.size()
    )
    {
        executionPlan.message =
            "Execution order does not contain every plan step.";

        return executionPlan;
    }

    // --------------------------------------------------------
    // STEP LOOKUP
    // --------------------------------------------------------

    unordered_map<string, const MultiStepAction*> stepMap;

    for (const auto& step : plan.steps)
    {
        if (step.id.empty())
        {
            executionPlan.message =
                "Cannot build execution plan from step with empty ID.";

            return executionPlan;
        }

        if (stepMap.contains(step.id))
        {
            executionPlan.message =
                "Duplicate step ID in execution plan: " +
                step.id;

            return executionPlan;
        }

        stepMap[step.id] =
            &step;
    }

    // --------------------------------------------------------
    // BUILD ORDERED EXECUTION STEPS
    // --------------------------------------------------------

    for (const string& stepId :
         executionOrder)
    {
        if (!stepMap.contains(stepId))
        {
            executionPlan.message =
                "Execution order references unknown step: " +
                stepId;

            executionPlan.steps.clear();

            return executionPlan;
        }

        const MultiStepAction& sourceStep =
            *stepMap[stepId];

        if (sourceStep.operations.empty())
        {
            executionPlan.message =
                "Step " +
                stepId +
                " contains no executable operations.";

            executionPlan.steps.clear();

            return executionPlan;
        }

        for (const PrimitiveOperation& operation :
             sourceStep.operations)
        {
            if (!operation.resolved)
            {
                executionPlan.message =
                    "Step " +
                    stepId +
                    " contains an unresolved primitive: " +
                    operation.name;

                executionPlan.steps.clear();

                return executionPlan;
            }

            ExecutionPlanStep executionStep;

            executionStep.id =
                stepId;

            executionStep.dependencies =
                sourceStep.dependencies;

            executionStep.operation =
                operation;

            executionStep.valid =
                true;

            executionPlan.steps.push_back(
                executionStep
            );
        }
    }

    // --------------------------------------------------------
    // FINAL VALIDATION
    // --------------------------------------------------------

    if (executionPlan.steps.empty())
    {
        executionPlan.message =
            "Execution plan contains no executable steps.";

        return executionPlan;
    }

    executionPlan.valid =
        true;

    executionPlan.message =
        "Multi-step execution plan built successfully.";

    return executionPlan;
}

#include "goal_decomposer.hpp"
#include "llm_client.hpp"

#include <nlohmann/json.hpp>

using namespace std;
using json = nlohmann::json;

MultiStepPlan GoalDecomposer::decompose(const string& goal)
{
    MultiStepPlan plan;

    plan.valid = false;
    plan.goal = goal;
    plan.steps.clear();
    plan.message = "";

    if (goal.empty())
    {
        plan.message = "Goal is empty.";
        return plan;
    }

    const string systemPrompt = R"(
You are Nova's Goal Decomposer.

Your ONLY responsibility is to decompose a user's high-level goal
into the smallest meaningful sequence of logical steps.

Do NOT:
- execute anything
- choose execution mechanisms
- invent capabilities
- resolve concrete system parameters
- produce shell commands
- produce code
- perform the requested actions
- decide whether Nova supports a capability

Each step must describe WHAT needs to happen, not HOW it will
be executed.

If one logical step may later require multiple basic operations,
keep it as one meaningful logical step.

Return STRICT JSON only.

Required format:

{
  "steps": [
    {
      "id": "step_1",
      "description": "..."
    },
    {
      "id": "step_2",
      "description": "...",
      "depends_on": ["step_1"]
    }
  ]
}

Rules:
- Step IDs must be unique.
- IDs must use the format step_1, step_2, step_3, etc.
- Keep steps in logical order.
- Use depends_on only when a step actually requires another step.
- Do not add unnecessary dependencies.
- Do not return markdown.
- Do not return explanations outside the JSON.

IMPORTANT DECOMPOSITION AND PRESERVATION RULES:

- Decompose ONLY what the user explicitly requested or what is
  logically unavoidable to fulfill that request.

- NEVER invent intermediate steps, resources, files, folders,
  directories, project structures, configurations, dependencies,
  or implementation conventions that the user did not request.

- Do not introduce standard project structures merely because they
  are conventional for a language, framework, IDE, or application.

- For example, if the user asks to create a Java project named
  "NovaJava" and add "Main.java", do NOT invent "src", "main",
  "java", Maven structure, Gradle structure, package directories,
  configuration files, or other project files unless the user
  explicitly requests them.

- Do not expand one requested action into additional implementation
  steps merely because those steps would commonly be used in practice.

- Every concrete user-provided name, filename, folder name,
  application name, path, location, or other explicit resource
  identifier MUST be preserved verbatim in the relevant step description.

- Do not replace concrete names with generic descriptions.

- Do not remove names such as "NovaJava", "Main.java", "Chrome", etc.

- Do not generalize a named resource into phrases such as
  "the project directory" when the concrete name is available.

- Preserve the user's requested level of abstraction.

- If the user requests one logical action, keep it as one logical
  step unless fulfilling it necessarily requires another explicitly
  requested logical action.

- Dependencies must represent actual logical prerequisites, not
  conventional workflow ordering.

- Step descriptions must contain enough information for deterministic
  downstream resolution of the information explicitly provided by
  the user.

- Do not infer additional user intent from domain conventions.

- The decomposer may restructure wording for clarity, but it must not
  add new user intent.)";

    const string userPrompt =
        "Decompose this user goal into logical steps:\n\n" + goal;

    LLMClient client;

    LLMResponse llmResponse = client.generate(
        systemPrompt,
        userPrompt
    );

    if (!llmResponse.success)
    {
        plan.message =
            "Goal decomposer LLM request failed: " +
            llmResponse.message;

        return plan;
    }

    string response = llmResponse.content;

    if (response.empty())
    {
        plan.message = "Goal decomposer received an empty LLM response.";
        return plan;
    }

    try
    {
        json parsed = json::parse(response);

        if (!parsed.contains("steps") ||
            !parsed["steps"].is_array())
        {
            plan.message =
                "Goal decomposer response does not contain a valid steps array.";

            return plan;
        }

        for (const auto& stepJson : parsed["steps"])
        {
            if (!stepJson.contains("id") ||
                !stepJson["id"].is_string())
            {
                plan.message =
                    "Goal decomposer returned a step without a valid id.";

                plan.steps.clear();
                return plan;
            }

            if (!stepJson.contains("description") ||
                !stepJson["description"].is_string())
            {
                plan.message =
                    "Goal decomposer returned a step without a valid description.";

                plan.steps.clear();
                return plan;
            }

            MultiStepAction step;

            step.id =
                stepJson["id"].get<string>();

            step.description =
                stepJson["description"].get<string>();

            step.operation = "";
            step.capability = "";
            step.parameters.clear();
            step.resolved = false;

            if (stepJson.contains("depends_on"))
            {
                if (!stepJson["depends_on"].is_array())
                {
                    plan.message =
                        "Goal decomposer returned an invalid depends_on field.";

                    plan.steps.clear();
                    return plan;
                }

                for (const auto& dependency :
                     stepJson["depends_on"])
                {
                    if (!dependency.is_string())
                    {
                        plan.message =
                            "Goal decomposer returned a non-string dependency.";

                        plan.steps.clear();
                        return plan;
                    }

                    step.dependencies.push_back(
                        dependency.get<string>()
                    );
                }
            }

            plan.steps.push_back(step);
        }

        if (plan.steps.empty())
        {
            plan.message =
                "Goal decomposer returned no steps.";

            return plan;
        }

        plan.valid = true;
        plan.message =
            "Goal decomposed successfully.";

        return plan;
    }
    catch (const json::exception& error)
    {
        plan.message =
            string("Invalid JSON from goal decomposer: ") +
            error.what();

        plan.steps.clear();

        return plan;
    }
}

{

    cout << "============================================================"
         << endl;
    cout << "NOVA"
         << endl;
    cout << "============================================================"
         << endl;

    cout << "[INPUT] Starting unified input pipeline..."
         << endl;

    // --------------------------------------------------------
    // PART A — WAKE WORD / LISTENING
    // --------------------------------------------------------

    PartABridge partABridge;

    PartAResult result =
        partABridge.waitForText();

    // --------------------------------------------------------
    // KEYBOARD OVERRIDE
    // --------------------------------------------------------

    if (
        result.success &&
        result.keyboardRequest
    )
    {
        cout << endl;
        cout << "[INPUT] Switching to keyboard input."
             << endl;

        KeyboardInput keyboardInput;

        InputEvent keyboardEvent =
            keyboardInput.waitForInput();

        return keyboardEvent;
    }

    // --------------------------------------------------------
    // SPEECH INPUT
    // --------------------------------------------------------

    if (
        result.success &&
        !result.keyboardRequest
    )
    {
        InputEvent event;

        event.source =
            InputSource::SPEECH;

        event.rawText =
            result.rawText;

        cout << endl;
        cout << "[INPUT] Speech input received."
             << endl;

        return event;
    }

    // --------------------------------------------------------
    // FAILURE
    // --------------------------------------------------------

    cout << endl;
    cout << "[INPUT] Input pipeline failed."
         << endl;

    InputEvent event;

    event.source =
        InputSource::SPEECH;

    event.rawText =
        "";

    return event;
}

NormalizedInput InputNormalizer::normalize(
    const string& input
)
{
    NormalizedInput result;

    result.originalText = input;
    result.normalizedText = "";
    result.success = false;
    result.message = "";

    if (input.empty())
    {
        result.message = "Cannot normalize empty input.";
        return result;
    }

    const string systemPrompt = R"(
You are Nova's Input Normalizer.

Your ONLY responsibility is to clean and normalize the user's input.

Do NOT:
- classify the request
- determine the user's intent
- select an action
- determine capabilities
- resolve parameters
- create an execution plan
- execute anything
- add information that was not present
- answer the user

You MUST:
- remove filler words and speech disfluencies
- remove unnecessary politeness and conversational padding
- remove "Nova" when it is only being used to address the assistant
- preserve the user's actual meaning
- preserve names
- preserve application names
- preserve file paths
- preserve numbers
- preserve quoted strings
- preserve requested operations
- preserve meaningful capitalization
- preserve all information required by later processing

Example:

Input:
"uhh Nova, can you like create a folder called Nova on my desktop"

Normalized:
"create a folder called Nova on my desktop"

If the input is already clean, return it without unnecessary changes.

Return ONLY valid JSON.

The JSON format MUST be exactly:

{
  "normalized_text": "..."
}

Do not return markdown.
Do not return code fences.
Do not return explanations.
Do not return any text outside the JSON object.
)";

    const string userPrompt =
        "Normalize this input:\n" + input;

    LLMClient llmClient;

    LLMResponse response =
        llmClient.generate(
            systemPrompt,
            userPrompt
        );

    if (!response.success)
    {
        result.message =
            "Input normalization LLM failed: "
            + response.message;

        return result;
    }

    try
    {
        json parsed =
            json::parse(response.content);

        if (!parsed.contains("normalized_text"))
        {
            result.message =
                "LLM response missing normalized_text.";

            return result;
        }

        if (!parsed["normalized_text"].is_string())
        {
            result.message =
                "LLM normalized_text is not a string.";

            return result;
        }

        result.normalizedText =
            parsed["normalized_text"].get<string>();

        if (result.normalizedText.empty())
        {
            result.message =
                "LLM returned empty normalized text.";

            return result;
        }

        result.success = true;
        result.message =
            "Input normalized successfully.";

        return result;
    }
    catch (const exception& exception)
    {
        result.message =
            "Failed to parse LLM normalization response: "
            + string(exception.what());

        return result;
    }
}

LLMResponse LLMClient::generate(
    const string& systemPrompt,
    const string& userPrompt
)
{
    LLMResponse result;

    result.success = false;
    result.content = "";
    result.message = "";

    // --------------------------------------------------------
    // API KEY
    // --------------------------------------------------------

    const char* apiKey =
        getenv("OPENROUTER_API_KEY");

    if (
        apiKey == nullptr ||
        string(apiKey).empty()
    )
    {
        result.message =
            "OPENROUTER_API_KEY environment variable is not set.";

        return result;
    }

    // --------------------------------------------------------
    // CURL INITIALIZATION
    // --------------------------------------------------------

    CURL* curl =
        curl_easy_init();

    if (curl == nullptr)
    {
        result.message =
            "Failed to initialize libcurl.";

        return result;
    }

    string responseBody;

    // --------------------------------------------------------
    // REQUEST BODY
    // --------------------------------------------------------

    json requestBody =
    {
        {
            "model",
            "dots-studio/dots-3-note-preview:free"
        },
        {
            "messages",
            json::array(
            {
                {
                    {
                        "role",
                        "system"
                    },
                    {
                        "content",
                        systemPrompt
                    }
                },
                {
                    {
                        "role",
                        "user"
                    },
                    {
                        "content",
                        userPrompt
                    }
                }
            })
        }
    };

    string requestBodyString =
        requestBody.dump();

    // --------------------------------------------------------
    // HEADERS
    // --------------------------------------------------------

    struct curl_slist* headers =
        nullptr;

    string authorization =
        "Authorization: Bearer "
        + string(apiKey);

    headers =
        curl_slist_append(
            headers,
            "Content-Type: application/json"
        );

    headers =
        curl_slist_append(
            headers,
            authorization.c_str()
        );

    // --------------------------------------------------------
    // CURL CONFIGURATION
    // --------------------------------------------------------

    curl_easy_setopt(
        curl,
        CURLOPT_URL,
        "https://openrouter.ai/api/v1/chat/completions"
    );

    curl_easy_setopt(
        curl,
        CURLOPT_HTTPHEADER,
        headers
    );

    curl_easy_setopt(
        curl,
        CURLOPT_POST,
        1L
    );

    curl_easy_setopt(
        curl,
        CURLOPT_POSTFIELDS,
        requestBodyString.c_str()
    );

    curl_easy_setopt(
        curl,
        CURLOPT_WRITEFUNCTION,
        writeCallback
    );

    curl_easy_setopt(
        curl,
        CURLOPT_WRITEDATA,
        &responseBody
    );

    curl_easy_setopt(
        curl,
        CURLOPT_CONNECTTIMEOUT,
        10L
    );

    curl_easy_setopt(
        curl,
        CURLOPT_TIMEOUT,
        60L
    );

    // --------------------------------------------------------
    // EXECUTE REQUEST
    // --------------------------------------------------------

    CURLcode curlResult =
        curl_easy_perform(curl);

    if (
        curlResult != CURLE_OK
    )
    {
        result.message =
            "OpenRouter request failed: "
            + string(
                curl_easy_strerror(
                    curlResult
                )
            );

        curl_slist_free_all(headers);
        curl_easy_cleanup(curl);

        return result;
    }

    // --------------------------------------------------------
    // HTTP STATUS
    // --------------------------------------------------------

    long httpStatus = 0;

    curl_easy_getinfo(
        curl,
        CURLINFO_RESPONSE_CODE,
        &httpStatus
    );

    if (
        httpStatus < 200 ||
        httpStatus >= 300
    )
    {
        result.message =
            "OpenRouter returned HTTP status "
            + to_string(httpStatus)
            + ".";

        cout
            << "[LLM] HTTP status: "
            << httpStatus
            << endl;

        cout
            << "[LLM] Response: "
            << responseBody
            << endl;

        curl_slist_free_all(headers);
        curl_easy_cleanup(curl);

        return result;
    }

    // --------------------------------------------------------
    // PARSE RESPONSE
    // --------------------------------------------------------

    try
    {
        json response =
            json::parse(
                responseBody
            );

        result.content =
            response
            ["choices"]
            [0]
            ["message"]
            ["content"]
            .get<string>();

        result.success = true;

        result.message =
            "LLM response received.";
    }
    catch (
        const exception& exception
    )
    {
        result.message =
            "Failed to parse OpenRouter response: "
            + string(
                exception.what()
            );
    }

    // --------------------------------------------------------
    // CLEANUP
    // --------------------------------------------------------

    curl_slist_free_all(headers);
    curl_easy_cleanup(curl);

    return result;
}

    const string& status

        chrono::system_clock::to_time_t(now);

    localtime_s(&localTime, &currentTime);
#else
    localtime_r(&currentTime, &localTime);
#endif

    ostringstream timestamp;

    timestamp << put_time(
        &localTime,
        "%Y-%m-%d %H:%M:%S"
    );

    string logEntry =
        "[" + timestamp.str() + "] "
        "[" + level + "] "
        "[" + event + "] "
        "[status=" + status + "] "
        + message;

    cout << logEntry << endl;

    logFileStream << logEntry << endl;
    logFileStream.flush();
}

#include "multi_step_executor.hpp"
#include "primitive_execution_engine.hpp"

#include <iostream>
#include <unordered_map>

using namespace std;

MultiStepExecutionResult MultiStepExecutor::execute(
    const ExecutionPlan& plan
)
{
    MultiStepExecutionResult result{};

    result.success = false;

    // --------------------------------------------------
    // Step 1: Validate execution plan
    // --------------------------------------------------

    if (!plan.valid)
    {
        result.message =
            "Multi-step execution plan is invalid.";

        return result;
    }

    if (plan.type != "multi_step")
    {
        result.message =
            "Execution plan is not a multi-step plan.";

        return result;
    }

    if (plan.steps.empty())
    {
        result.message =
            "Multi-step execution plan contains no steps.";

        return result;
    }

    // --------------------------------------------------
    // Step 2: Track completed steps
    // --------------------------------------------------

    unordered_map<string, bool> completedSteps;

    PrimitiveExecutionEngine primitiveEngine;

    // --------------------------------------------------
    // Step 3: Execute steps in execution-plan order
    // --------------------------------------------------

    for (const ExecutionPlanStep& step : plan.steps)
    {
        cout
            << "[MULTI-STEP EXECUTION] "
            << step.id
            << endl;

        // ----------------------------------------------
        // Validate current step
        // ----------------------------------------------

        if (!step.valid)
        {
            result.message =
                "Step is invalid: "
                + step.id;

            return result;
        }

        // ----------------------------------------------
        // Check dependencies
        // ----------------------------------------------

        for (const string& dependency : step.dependencies)
        {
            auto dependencyResult =
                completedSteps.find(dependency);

            if (
                dependencyResult ==
                completedSteps.end()
                ||
                !dependencyResult->second
            )
            {
                result.message =
                    "Dependency was not successfully completed for step: "
                    + step.id;

                return result;
            }
        }

        // ----------------------------------------------
        // Execute primitive
        // ----------------------------------------------
        cout
            << "[PRIMITIVE PARAMETERS]"
            << endl;

        cout
            << "  Name: "
            << step.operation.parameters.name
            << endl;

        cout
            << "  Location: "
            << step.operation.parameters.location
            << endl;

        cout
            << "  Application: "
            << step.operation.parameters.application
            << endl;

        cout
            << "  Target: "
            << step.operation.parameters.target
            << endl;

        cout
            << "  Content: "
            << step.operation.parameters.content
            << endl;

        ExecutionResult executionResult =
            primitiveEngine.execute(
                step.operation
            );

        result.stepResults.push_back(
            executionResult
        );

        cout
            << "[STEP RESULT] "
            << executionResult.message
            << endl;

        // ----------------------------------------------
        // Stop immediately on failure
        // ----------------------------------------------

        if (
            executionResult.status
            !=
            ExecutionStatus::SUCCESS
        )
        {
            completedSteps[step.id] = false;

            result.message =
                "Multi-step execution stopped at step: "
                + step.id;

            return result;
        }

        completedSteps[step.id] = true;
    }

    // --------------------------------------------------
    // Step 4: All steps completed
    // --------------------------------------------------

    result.success = true;

    result.message =
        "All multi-step operations executed successfully.";

    return result;
}

MultiStepPipelineResult MultiStepPipeline::run(
    const string& goal
)
{
    MultiStepPipelineResult result;

    result.status =
        MultiStepPipelineStatus::FAILED;

    // ============================================================
    // 1. GOAL DECOMPOSITION
    // ============================================================

    GoalDecomposer goalDecomposer;

    MultiStepPlan plan =
        goalDecomposer.decompose(
            goal
        );

    if (!plan.valid)
    {
        result.message =
            "Goal decomposition failed: " +
            plan.message;

        return result;
    }
    cout << "\n[DECOMPOSED STEPS]\n";
    for (const MultiStepAction& step : plan.steps)
    {
        cout << "Step ID: "
            << step.id
            << endl;

        cout << "Description: "
            << step.description
            << endl;

        cout << "Dependencies: ";

        for (const string& dependency : step.dependencies)
        {
            cout << dependency << " ";
        }

        cout << endl;

        cout << "------------------------"
            << endl;
    }

    // ============================================================
    // 2. DEPENDENCY RESOLUTION
    // ============================================================

    DependencyResolver dependencyResolver;

    DependencyResolutionResult dependencyResult =
        dependencyResolver.resolve(
            plan
        );

    if (!dependencyResult.valid)
    {
        result.message =
            "Dependency resolution failed: " +
            dependencyResult.message;

        return result;
    }
    plan =
        dependencyResult.plan;

    // ============================================================
    // 3. ACTION / PRIMITIVE RESOLUTION
    // ============================================================

    ActionPrimitiveResolver actionPrimitiveResolver;

    for (MultiStepAction& step : plan.steps)
    {
        ActionPrimitiveResolutionResult primitiveResult =
            actionPrimitiveResolver.resolve(
                step
            );

        if (
            primitiveResult.status !=
            ActionPrimitiveResolutionStatus::RESOLVED
        )
        {
            result.message =
                "Action / primitive resolution failed for " +
                step.id +
                ": " +
                primitiveResult.message;

            if (
                primitiveResult.status ==
                ActionPrimitiveResolutionStatus::CLARIFICATION_REQUIRED
            )
            {
                result.message +=
                    " Clarification: " +
                    primitiveResult.clarificationQuestion;
            }

            return result;
        }

                step =
                    primitiveResult.action;

                cout << "\n[PRIMITIVE RESOLUTION] "
                    << step.id
                    << endl;

                for (
                    const PrimitiveOperation& operation
                    : step.operations
                )
                {
                    cout << "  Primitive: "
                        << operation.name
                        << endl;

                    cout << "  Resolved: "
                        << (
                            operation.resolved
                            ? "true"
                            : "false"
                        )
                        << endl;
                }
            }
    // ============================================================
    // 4. CAPABILITY + PARAMETER RESOLUTION
    // ============================================================

    CapabilityResolver capabilityResolver;

    ParameterResolver parameterResolver;

    for (MultiStepAction& step : plan.steps)
    {
        step.semanticParameters.clear();
        for (PrimitiveOperation& operation : step.operations)
        {
            Capability capability =
                capabilityResolver.resolve(
                    operation
                );

            if (!capability.supported)
            {
                result.message =
                    "Unsupported capability for " +
                    operation.name +
                    ": " +
                    capability.message;

                return result;
            }

            vector<pair<string, string>> availableContext;
            for (const MultiStepAction& previousStep : plan.steps)
            {
                if (find(
                        step.dependencies.begin(),
                        step.dependencies.end(),
                        previousStep.id) != step.dependencies.end())
                {
                    availableContext.emplace_back(
                        previousStep.id,
                        previousStep.description);
                }
            }

            ParameterResolverResult parameterResult =
                parameterResolver.resolve(
                    step.description,
                    operation,
                    capability,
                    availableContext
                );

            if (
                parameterResult.status ==
                    ParameterResolutionStatus::AMBIGUOUS ||
                parameterResult.status ==
                    ParameterResolutionStatus::MISSING_REQUIRED_PARAMETER
            )
            {
                result.status =
                    MultiStepPipelineStatus::CLARIFICATION_REQUIRED;

                result.message =
                    "Parameter resolution requires clarification for " +
                    step.id +
                    ": " +
                    parameterResult.message;

                result.clarificationQuestion =
                    parameterResult.clarificationQuestion;

                return result;
            }

            if (!parameterResult.success)
            {
                result.status =
                    MultiStepPipelineStatus::FAILED;

                result.message =
                    "Parameter resolution failed for " +
                    operation.name +
                    " in " +
                    step.id +
                    ": " +
                    parameterResult.message;

                return result;
            }

            step.semanticParameters.push_back(
                parameterResult.parameters);
        }
    }

    // ============================================================
    // 5. CONTEXT RESOLUTION
    // ============================================================

    ContextResolver contextResolver;

    for (size_t i = 0; i < plan.steps.size(); ++i)
    {
        MultiStepAction& step =
            plan.steps[i];

        MultiStepPlan contextPlan =
            plan;

        ContextResolutionResult contextResult =
            contextResolver.resolve(
                step,
                contextPlan
            );

        if (
            contextResult.status ==
            ContextResolutionStatus::CLARIFICATION_REQUIRED
        )
        {
            result.message =
                "Context resolution requires clarification: " +
                contextResult.clarificationQuestion;

            return result;
        }

        if (
            contextResult.status ==
            ContextResolutionStatus::FAILED
        )
        {
            result.message =
                "Context resolution failed for " +
                step.id +
                ": " +
                contextResult.message;

            return result;
        }

        step =
            contextResult.action;
    }

    // ============================================================
    // 6. FINAL RESULT
    // ============================================================

    plan.valid =
        true;

    result.status =
        MultiStepPipelineStatus::SUCCESS;

    PlanValidator planValidator;

    PlanValidationResult validationResult =
        planValidator.validate(plan);

    if (validationResult.status ==
        PlanValidationStatus::INVALID)
    {
        result.status =
            MultiStepPipelineStatus::FAILED;

        result.message =
            "Plan validation failed: " +
            validationResult.message;

        return result;
    }

    cout << "\n[PLAN VALIDATION]\n";
    cout << "Status: VALID" << endl;
    cout << "Message: "
        << validationResult.message
        << endl;

    // --------------------------------------------------
    // EXECUTION PLAN
    // --------------------------------------------------

    ExecutionPlanBuilder executionPlanBuilder;

    ExecutionPlan executionPlan =
        executionPlanBuilder.build(

            dependencyResult.executionOrder
        );

            << endl;

        cout << "------------------------"
            << endl;

namespace

    string stripWrappingQuotes(const string& value)
    {
        if (value.size() >= 2)
        {
            const char first = value.front();
            const char last = value.back();

            if ((first == '"' && last == '"') ||
                (first == '\'' && last == '\''))
            {
                return value.substr(1, value.size() - 2);
            }
        }

        return value;
    }

    bool hasExactlyKeys(
        const json& value,
        initializer_list<const char*> keys
    )
    {
        if (!value.is_object() || value.size() != keys.size())
        {
            return false;
        }

        for (const char* key : keys)
        {
            if (!value.contains(key))
            {
                return false;
            }
        }

        return true;
    }

    bool isMeaningful(const string& value)
    {
        return any_of(
            value.begin(),
            value.end(),
            [](unsigned char character)
            {
                return !isspace(character);
            }
        );
    }

    const char* requirementName(ParameterRequirement requirement)
    {
        switch (requirement)
        {
            case ParameterRequirement::NOT_APPLICABLE:
                return "not_applicable";
            case ParameterRequirement::REQUIRED:
                return "required";
            case ParameterRequirement::OPTIONAL:
                return "optional";
        }

        return "invalid";
    }

    bool parseParameterValue(
        const json& value,
        ParameterValue& parameter
    )
    {
        if (!value.is_object() || !value.contains("type") ||
            !value["type"].is_string())
        {
            return false;
        }

        const string type = value["type"].get<string>();

        if (type == "none")
        {
            if (!hasExactlyKeys(value, {"type"}))
            {
                return false;
            }

            parameter = ParameterValue{};
            return true;
        }

        if (type == "literal")
        {
            if (!hasExactlyKeys(value, {"type", "value"}) ||
                !value["value"].is_string())
            {
                return false;
            }

            parameter.type = ParameterValueType::LITERAL;
            parameter.value = value["value"].get<string>();
            parameter.reference = ContextParameterReference{};
            return true;
        }

        if (type == "context_reference")
        {
            if (!hasExactlyKeys(value, {"type", "step_id", "output"}) ||
                !value["step_id"].is_string() ||
                !value["output"].is_string())
            {
                return false;
            }

            parameter.type = ParameterValueType::CONTEXT_REFERENCE;
            parameter.value.clear();
            parameter.reference.stepId =
                value["step_id"].get<string>();
            parameter.reference.output =
                value["output"].get<string>();
            return true;
        }

        return false;
    }

    bool parameterMatchesRequirement(
        const ParameterValue& parameter,
        ParameterRequirement requirement,
        const string& name,
        string& errorMessage
    )
    {
        if (parameter.type == ParameterValueType::NONE)
        {
            if (requirement == ParameterRequirement::REQUIRED)
            {
                errorMessage = "Required parameter is missing: " + name;
                return false;
            }

            return true;
        }

        if (requirement == ParameterRequirement::NOT_APPLICABLE)
        {
            errorMessage = "Parameter is not applicable: " + name;
            return false;
        }

        if (parameter.type == ParameterValueType::LITERAL)
        {
            if (!isMeaningful(parameter.value))
            {
                errorMessage = "Literal parameter is empty: " + name;
                return false;
            }

            return true;
        }

        if (parameter.type == ParameterValueType::CONTEXT_REFERENCE)
        {
            if (!isMeaningful(parameter.reference.stepId) ||
                !isMeaningful(parameter.reference.output))
            {
                errorMessage = "Context reference is incomplete: " + name;
                return false;
            }

            return true;
        }

        errorMessage = "Unknown parameter value type: " + name;
        return false;
    }

    string makeSystemPrompt(
        const PrimitiveDefinition& definition,
        const PrimitiveParameterSchema& schema
    )
    {
        ostringstream prompt;
        prompt << R"(You are Nova's semantic parameter resolver.

Each parameter must be exactly one of:

{"type":"literal","value":"..."}

Ambiguous response:

Rules:

        prompt << "\n\nPrimitive type: "
               << static_cast<int>(definition.type)
               << "\nParameter requirements:\n"
               << "name: " << requirementName(schema.name) << "\n"
               << "location: " << requirementName(schema.location) << "\n"
               << "application: " << requirementName(schema.application) << "\n"
               << "target: " << requirementName(schema.target) << "\n"
               << "content: " << requirementName(schema.content);
        return prompt.str();
    }

    string makeUserPrompt(
        const string& description,
        const PrimitiveOperation& operation,
        const vector<pair<string, string>>& availableContext
    )
    {

        prompt << "Primitive: " << operation.name
            << "\nStep description: " << description

            << "\nValid context outputs are path, name, location, application, target, and content."

bool ParameterResolver::validateParameters(
    const ActionParameters& parameters,
    const PrimitiveParameterSchema& schema,
    string& errorMessage

        parameterMatchesRequirement(parameters.name, schema.name, "name", errorMessage) &&

ParameterResolverResult ParameterResolver::resolve(
    const string& input,
    const Action& action,
    const Capability& capability
)
{
    ParameterResolverResult result{};
    result.status = ParameterResolutionStatus::FAILED;

    if (input.empty())
    {
        result.message = "Cannot resolve parameters from empty input.";
        return result;
    }

    if (!capability.supported)
    {
        result.message =
            "Cannot resolve parameters for unsupported capability.";
        return result;
    }

    if (action.name == "create_folder")
    {
        const size_t marker = input.find("called ");
        if (marker == string::npos)
        {
            result.status = ParameterResolutionStatus::MISSING_REQUIRED_PARAMETER;
            result.message = "Folder name is required.";
            result.clarificationQuestion =
                "What should the project folder be named?";
            return result;
        }

        const size_t nameStart = marker + string("called ").size();
        size_t nameEnd = input.find(" on ", nameStart);
        if (nameEnd == string::npos)
        {
            nameEnd = input.size();
        }

        const string name = stripWrappingQuotes(
            input.substr(nameStart, nameEnd - nameStart));
        if (!isMeaningful(name))
        {
            result.status = ParameterResolutionStatus::MISSING_REQUIRED_PARAMETER;
            result.message = "Folder name is required.";
            result.clarificationQuestion =
                "What should the project folder be named?";
            return result;
        }

        result.parameters.name = makeLiteral(name);
        result.parameters.location = makeLiteral("desktop");
        result.success = true;
        result.status = ParameterResolutionStatus::RESOLVED;
        result.message = "Folder parameters resolved successfully.";
        return result;
    }

    if (action.name == "open_app" || action.name == "close_app")
    {
        string application;
        const string prefixes[] = {"open ", "launch ", "close "};
        for (const string& prefix : prefixes)
        {
            if (input.rfind(prefix, 0) == 0)
            {
                application = stripWrappingQuotes(input.substr(prefix.size()));
                break;
            }
        }

        if (!isMeaningful(application))
        {
            result.status = ParameterResolutionStatus::MISSING_REQUIRED_PARAMETER;
            result.message = "Application parameter is required.";
            result.clarificationQuestion =
                "Which application should Nova use?";
            return result;
        }

        result.parameters.application = makeLiteral(application);
        result.success = true;
        result.status = ParameterResolutionStatus::RESOLVED;
        result.message = "Application parameter resolved successfully.";
        return result;
    }

    result.message = "No parameter resolution rule exists for requested action.";
    return result;
}

ParameterResolverResult ParameterResolver::resolveSemanticParameters(
    const string& description,
    const PrimitiveOperation& operation,
    const PrimitiveDefinition& definition,
    const vector<pair<string, string>>& availableContext
)
{
    ParameterResolverResult result{};
    result.status = ParameterResolutionStatus::FAILED;

    const PrimitiveParameterSchema& schema = *definition.parameters;
    const string systemPrompt = makeSystemPrompt(definition, schema);
    const string userPrompt = makeUserPrompt(
        description,
        operation,
        availableContext);
    LLMResponse response;

    if (llmGenerator)
    {
        response = llmGenerator(systemPrompt, userPrompt);
    }
    else
    {
        LLMClient client;
        response = client.generate(systemPrompt, userPrompt);
    }

    if (!response.success || response.content.empty())
    {
        result.message = "Parameter resolution LLM request failed: " +
            response.message;
        return result;
    }

    try
    {
        const json parsed = json::parse(response.content);
        if (!parsed.is_object() || !parsed.contains("status") ||
            !parsed["status"].is_string())
        {
            result.status = ParameterResolutionStatus::INVALID_RESPONSE;
            result.message = "Parameter resolver response has no valid status.";
            return result;
        }

        const string status = parsed["status"].get<string>();
        if (status == "ambiguous" ||
            status == "missing_required_parameter")
        {
            if (!hasExactlyKeys(parsed, {"status", "question"}) ||
                !parsed["question"].is_string() ||
                !isMeaningful(parsed["question"].get<string>()))
            {
                result.status = ParameterResolutionStatus::INVALID_RESPONSE;
                result.message =
                    "Parameter resolver clarification response is malformed.";
                return result;
            }

            result.status = status == "ambiguous"
                ? ParameterResolutionStatus::AMBIGUOUS
                : ParameterResolutionStatus::MISSING_REQUIRED_PARAMETER;
            result.message = status == "ambiguous"
                ? "Parameter resolution is ambiguous."
                : "A required parameter is missing.";
            result.clarificationQuestion =
                parsed["question"].get<string>();
            return result;
        }

        if (status != "resolved" ||
            !hasExactlyKeys(parsed, {"status", "parameters"}) ||
            !hasExactlyKeys(
                parsed["parameters"],
                {"name", "location", "application", "target", "content"}))
        {
            result.status = ParameterResolutionStatus::INVALID_RESPONSE;
            result.message = "Parameter resolver returned an invalid JSON schema.";
            return result;
        }

        const json& parameters = parsed["parameters"];
        const bool parsedValues =
            parseParameterValue(parameters["name"], result.parameters.name) &&
            parseParameterValue(parameters["location"], result.parameters.location) &&
            parseParameterValue(parameters["application"], result.parameters.application) &&
            parseParameterValue(parameters["target"], result.parameters.target) &&
            parseParameterValue(parameters["content"], result.parameters.content);

        if (!parsedValues)
        {
            result.status = ParameterResolutionStatus::INVALID_RESPONSE;
            result.message = "Parameter resolver returned an invalid parameter value.";
            return result;
        }

        string validationError;
        if (!validateParameters(result.parameters, schema, validationError))
        {
            result.status = ParameterResolutionStatus::INVALID_RESPONSE;
            result.message = "Parameter schema validation failed: " +
                validationError;
            return result;
        }

        result.success = true;
        result.status = ParameterResolutionStatus::RESOLVED;
        result.message = "Semantic parameters resolved successfully.";
        return result;
    }
    catch (const exception& error)
    {
        result.status = ParameterResolutionStatus::INVALID_RESPONSE;
        result.message = string("Invalid parameter resolver JSON: ") +
            error.what();
        return result;
    }
}

ParameterResolverResult ParameterResolver::resolve(
    const string& stepDescription,
    const PrimitiveOperation& operation,
    const Capability& capability,
    const vector<pair<string, string>>& availableContext
)
{
    ParameterResolverResult result{};
    result.status = ParameterResolutionStatus::FAILED;

    if (stepDescription.empty())
    {
        result.message =
            "Cannot resolve parameters from empty step description.";
        return result;
    }

    if (!operation.resolved)
    {
        result.message =
            "Cannot resolve parameters for unresolved primitive.";
        return result;
    }

    if (!capability.supported)
    {
        result.message =
            "Cannot resolve parameters for unsupported capability.";
        return result;
    }

    PrimitiveDefinition definition{};
    if (!PrimitiveRegistry::resolveDefinition(operation.name, definition))
    {
        result.message = "Unsupported primitive: " + operation.name;
        return result;
    }

    if (operation.type != definition.type)
    {
        result.status = ParameterResolutionStatus::INVALID_RESPONSE;
        result.message = "Primitive name and operation type do not match.";
        return result;
    }

    if (!definition.parameters.has_value())
    {
        result.message =
            "Primitive parameter schema is not defined: " + operation.name;
        return result;
    }

    return resolveSemanticParameters(
        stepDescription,
        operation,
        definition,
        availableContext);
}

bool ParameterResolver::toPrimitiveParameters(
    const ActionParameters& parameters,
    PrimitiveParameters& primitiveParameters,
    string& errorMessage
)
{
    const auto convert = [&errorMessage](
        const ParameterValue& value,
        string& output,
        const string& name)
    {
        if (value.type == ParameterValueType::NONE)
        {
            output.clear();
            return true;
        }

        if (value.type != ParameterValueType::LITERAL ||
            !isMeaningful(value.value))
        {
            errorMessage =
                "Parameter must be resolved to a meaningful literal before execution: " +
                name;
            return false;
        }

        output = value.value;
        return true;
    };

    return
        convert(parameters.name, primitiveParameters.name, "name") &&
        convert(parameters.location, primitiveParameters.location, "location") &&
        convert(parameters.application, primitiveParameters.application, "application") &&
        convert(parameters.target, primitiveParameters.target, "target") &&
        convert(parameters.content, primitiveParameters.content, "content");
}

bool ParameterResolver::resolveIntoOperation(
    const string& stepDescription,
    PrimitiveOperation& operation,
    const Capability& capability,
    const vector<pair<string, string>>& availableContext
)
{
    const ParameterResolverResult result = resolve(
        stepDescription,
        operation,
        capability,
        availableContext);

    if (!result.success)
    {
        return false;
    }

    PrimitiveParameters primitiveParameters;
    string errorMessage;
    if (!toPrimitiveParameters(
            result.parameters,
            primitiveParameters,
            errorMessage))
    {
        return false;
    }

    operation.parameters = primitiveParameters;
    return true;
}

PlanValidationResult PlanValidator::validate(
    const MultiStepPlan& plan
)
{
    PlanValidationResult result;

    result.status =
        PlanValidationStatus::INVALID;

    if (!plan.valid)
    {
        result.message =
            "Plan is marked invalid.";

        return result;
    }

    if (plan.steps.empty())
    {
        result.message =
            "Plan contains no steps.";

        return result;
    }

    for (const MultiStepAction& step : plan.steps)
    {
        if (step.id.empty())
        {
            result.message =
                "Plan contains a step with no ID.";

            return result;
        }

        if (step.operations.empty())
        {
            result.message =
                "Step " + step.id +
                " contains no operations.";

            return result;
        }

        for (const PrimitiveOperation& operation :
             step.operations)
        {
            if (!operation.resolved)
            {
                result.message =
                    "Unresolved primitive in step " +
                    step.id + ": " +
                    operation.name;

                return result;
            }
        }

        for (const ContextReference& reference :
             step.contextReferences)
        {
            if (!reference.resolved)
            {
                result.message =
                    "Unresolved context reference in step " +
                    step.id;

                return result;
            }
        }
    }

    result.status =
        PlanValidationStatus::VALID;

    result.message =
        "Plan passed structural validation.";

    return result;
}

ExecutionResult PrimitiveExecutionEngine::execute(
    const PrimitiveOperation& operation
)
{
    ExecutionResult result{};

    result.action = operation.name;
    result.confidence = 0.0;
    result.targetResolved = false;
    result.executionMechanismResolved = false;

    // --------------------------------------------------
    // Step 1: Resolve primitive
    // --------------------------------------------------

    if (!operation.resolved)
    {
        result.status =
            ExecutionStatus::NOT_EXECUTED;

        result.reason =
            ExecutionFailureReason::INVALID_PLAN;

        result.message =
            "Primitive operation is unresolved.";

        return result;
    }

    // --------------------------------------------------
    // Step 2: filesystem.create
    // --------------------------------------------------

    if (
        operation.type ==
            PrimitiveOperationType::FILESYSTEM_CREATE
        &&
        operation.name ==
            "filesystem.create"
    )
    {
        const string& name =
            operation.parameters.name;

        const string& location =
            operation.parameters.location;

        if (
            name.empty()
            ||
            location.empty()
        )
        {
            result.status =
                ExecutionStatus::NOT_EXECUTED;

            result.reason =
                ExecutionFailureReason::TARGET_NOT_FOUND;

            result.message =
                "Folder name or location is missing.";

            return result;
        }

        if (location != "desktop")
        {
            result.status =
                ExecutionStatus::NOT_EXECUTED;

            result.reason =
                ExecutionFailureReason::UNSUPPORTED_ACTION;

            result.message =
                "Unsupported folder location.";

            return result;
        }

        filesystem::path desktopPath =
            getDesktopPath();

        if (desktopPath.empty())
        {
            result.status =
                ExecutionStatus::FAILED;

            result.reason =
                ExecutionFailureReason::TARGET_NOT_FOUND;

            result.message =
                "Unable to resolve Windows Desktop location.";

            return result;
        }

        filesystem::path folderPath =
            desktopPath / name;

        result.targetResolved = true;
        result.executionMechanismResolved = true;

        result.confidence = 1.0;

        // --------------------------------------------------
        // Step 3: Execute
        // --------------------------------------------------

        try
        {
            if (filesystem::exists(folderPath))
            {
                result.status =
                    ExecutionStatus::FAILED;

                result.reason =
                    ExecutionFailureReason::EXECUTION_ERROR;

                result.message =
                    "Folder already exists: "
                    + folderPath.string();

                return result;
            }

            if (
                filesystem::create_directories(
                    folderPath
                )
            )
            {
                result.status =
                    ExecutionStatus::SUCCESS;

                result.reason =
                    ExecutionFailureReason::NONE;

                result.message =
                    "Folder created: "
                    + folderPath.string();

                return result;
            }
        }
        catch (
            const filesystem::filesystem_error& error
        )
        {
            result.status =
                ExecutionStatus::FAILED;

            result.reason =
                ExecutionFailureReason::EXECUTION_ERROR;

            result.message =
                error.what();

            return result;
        }

        result.status =
            ExecutionStatus::FAILED;

        result.reason =
            ExecutionFailureReason::EXECUTION_ERROR;

        result.message =
            "Folder could not be created.";

        return result;
    }

    // --------------------------------------------------
    // Unsupported primitive
    // --------------------------------------------------

    result.status =
        ExecutionStatus::NOT_EXECUTED;

    result.reason =
        ExecutionFailureReason::UNSUPPORTED_ACTION;

    result.message =
        "No execution mechanism exists for primitive: "
        + operation.name;

    return result;
}

bool PrimitiveRegistry::resolveDefinition(
    const string& primitiveName,
    PrimitiveDefinition& definition
)
{
    auto iterator =
        PRIMITIVE_REGISTRY.find(
            primitiveName
        );

    if (iterator == PRIMITIVE_REGISTRY.end())
    {
        return false;
    }

    definition = iterator->second;

    return true;
}

    vector<string> names;

    for (const auto& entry :

        names.push_back(entry.first);
    }

    return names;

        RequestType requestType;

        if (!stringToRequestType(
                classification,
                requestType
            ))
        {
            result.message =
                "LLM returned unknown classification: "
                + classification;

            return result;
        }

        result.type = requestType;
        result.success = true;
        result.message =
            "Request classified successfully.";

        return result;
    }
    catch (const exception& exception)
    {
        result.message =
            "Failed to parse Root Router response: "
            + string(exception.what());

        return result;
    }
}

#include "execution_engine.hpp"

using namespace std;

SingleActionResult SingleActionPipeline::process(
    const string& input
)
{
    SingleActionResult result;

    result.success = false;
    result.executionResult.status =
        ExecutionStatus::NOT_EXECUTED;

    result.executionResult.reason =
        ExecutionFailureReason::INVALID_PLAN;

    result.executionResult.action = "";

    result.executionResult.message =
        "Execution has not started.";

    result.executionResult.confidence = 0.0;
    result.executionResult.targetResolved = false;
    result.executionResult.executionMechanismResolved = false;

    result.action.name = "";

    result.capability.supported = false;
    result.capability.name = "";
    result.capability.message = "";

    result.parameters.name = ParameterValue{};
    result.parameters.location = ParameterValue{};
    result.parameters.application = ParameterValue{};

    result.validation.valid = false;
    result.validation.message = "";

    result.executionPlan.valid = false;
    result.executionPlan.type = "";
    result.executionPlan.action.name = "";
    result.executionPlan.capability.supported = false;
    result.executionPlan.capability.name = "";
    result.executionPlan.capability.message = "";
    result.executionPlan.parameters = PrimitiveParameters{};
    result.executionPlan.message = "";

    result.message = "";

    // --------------------------------------------------------
    // STAGE 1 — ACTION PARSER
    // --------------------------------------------------------

    ActionParser actionParser;

    ActionParserResult actionResult =
        actionParser.parse(input);

    if (!actionResult.success)
    {
        result.message =
            actionResult.message;

        return result;
    }

    result.action =
        actionResult.action;

    // --------------------------------------------------------
    // STAGE 2 — CAPABILITY RESOLVER
    // --------------------------------------------------------

    CapabilityResolver capabilityResolver;

    Capability capability =
        capabilityResolver.resolve(
            result.action
        );

    if (!capability.supported)
    {
        result.message =
            capability.message;

        return result;
    }

    result.capability =
        capability;

    // --------------------------------------------------------
    // STAGE 3 — PARAMETER RESOLVER
    // --------------------------------------------------------

    ParameterResolver parameterResolver;

    ParameterResolverResult parameterResult =
        parameterResolver.resolve(
            input,
            result.action,
            result.capability
        );

    if (!parameterResult.success)
    {
        result.message =
            parameterResult.message;

        return result;
    }

    result.parameters =
        parameterResult.parameters;

    // --------------------------------------------------------
    // STAGE 4 — ACTION VALIDATOR
    // --------------------------------------------------------

    ActionValidator actionValidator;

    ActionValidationResult validationResult =
        actionValidator.validate(
            result.action,
            result.capability,
            result.parameters
        );

    result.validation =
        validationResult;

    if (!validationResult.valid)
    {
        result.message =
            validationResult.message;

        return result;
    }

    // --------------------------------------------------------
    // STAGE 5 — EXECUTION PLAN
    // --------------------------------------------------------

    ExecutionPlanBuilder executionPlanBuilder;

    ExecutionPlan executionPlan =
        executionPlanBuilder.build(
            result.action,
            result.capability,
            result.parameters,
            result.validation
        );

    result.executionPlan =
        executionPlan;

    if (!executionPlan.valid)
    {
        result.message =
            executionPlan.message;

        return result;
        }
    // --------------------------------------------------------
    // STAGE 6 — CONFIDENCE & EXECUTION
    // --------------------------------------------------------

    ExecutionEngine executionEngine;

    ExecutionResult executionResult =
        executionEngine.execute(
            result.executionPlan
        );

    result.executionResult =
        executionResult;

    if (
        executionResult.status
        !=
        ExecutionStatus::SUCCESS
    )
    {
        result.message =
            executionResult.message;

        return result;
    }
    // --------------------------------------------------------
    // PIPELINE SUCCESS
    // --------------------------------------------------------

    result.success = true;

    result.message =
        "Single action executed successfully.";
    return result;
}
